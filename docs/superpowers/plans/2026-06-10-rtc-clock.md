# RTC + Clock Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A real 12-hour clock in the reader (from the PCF85063 RTC) plus a "Set clock" screen.

**Architecture:** Pure conversions (BCD, 12h formatting) live in host-tested `core/rtctime`. A new C++ `rtc_bsp` component does the I2C read/write on the existing touch bus. `ui_reader` shows the time on a timer; `ui_menu` adds a Set-clock screen. The reader stays usable throughout.

**Tech Stack:** ESP-IDF 5.5.2 (`esp_driver_i2c`), LVGL 9, C++17 `core` (doctest).

**PCF85063:** addr `0x51` on `user_i2c_port1_handle` (I2C_NUM_1). Regs: Control_1 `0x00` (bit1 `12_24`: 0=24h), Seconds `0x04` (bit7 OS, bits6-0 BCD), Minutes `0x05`, Hours `0x06` (BCD 24h).

```cpp
#include "rsvp/rtctime.hpp"
#include <cstdio>

namespace rsvp {

std::uint8_t bcdToBin(std::uint8_t bcd) { return (std::uint8_t)((bcd >> 4) * 10 + (bcd & 0x0F)); }
std::uint8_t binToBcd(std::uint8_t bin) { return (std::uint8_t)(((bin / 10) << 4) | (bin % 10)); }

std::string formatClock12h(std::uint8_t hour24, std::uint8_t minute) {
    int h = hour24 % 24;
    const char* ap = (h < 12) ? "AM" : "PM";
    int h12 = h % 12; if (h12 == 0) h12 = 12;
    char buf[16];
    std::snprintf(buf, sizeof buf, "%d:%02d %s", h12, minute % 60, ap);
    return buf;
}

std::uint8_t to24h(std::uint8_t hour12, bool isPM) {
    int h = hour12 % 12;       // 12 -> 0
    if (isPM) h += 12;
    return (std::uint8_t)h;
}

} // namespace rsvp
```

- [ ] **Step 2: Implementation** — `firmware/components/rtc_bsp/rtc_bsp.cpp`:
```cpp
#include "rtc_bsp.hpp"
#include "i2c_bsp.h"               // user_i2c_port1_handle, rtc_dev_handle
#include "driver/i2c_master.h"
#include "esp_log.h"

using rsvp::RtcTime;
static const char* TAG = "rtc_bsp";
static const int   kTimeoutMs = 1000;
#define PCF85063_ADDR 0x51

void rtc_init() {
    if (rtc_dev_handle) return;
    if (!user_i2c_port1_handle) { ESP_LOGW(TAG, "touch I2C bus not ready"); return; }
    i2c_device_config_t dev = { .dev_addr_length = I2C_ADDR_BIT_LEN_7,
                                .device_address = PCF85063_ADDR, .scl_speed_hz = 100000 };
    if (i2c_master_bus_add_device(user_i2c_port1_handle, &dev, &rtc_dev_handle) != ESP_OK) {
        ESP_LOGW(TAG, "add PCF85063 failed"); rtc_dev_handle = NULL; return;
    }
    uint8_t ctrl1[2] = { 0x00, 0x00 };   // Control_1 = 24h mode, running
    i2c_master_transmit(rtc_dev_handle, ctrl1, 2, kTimeoutMs);
    ESP_LOGI(TAG, "PCF85063 ready");
}

static bool read_regs(uint8_t start, uint8_t* buf, size_t n) {
    if (!rtc_dev_handle) return false;
    return i2c_master_transmit_receive(rtc_dev_handle, &start, 1, buf, n, kTimeoutMs) == ESP_OK;
}

bool rtc_valid() {
    uint8_t sec = 0;
    if (!read_regs(0x04, &sec, 1)) return false;
    return (sec & 0x80) == 0;            // OS flag clear -> time is valid
}

bool rtc_get(RtcTime& out) {
    uint8_t b[3] = {0};                  // sec(0x04), min(0x05), hour(0x06)
    if (!read_regs(0x04, b, 3)) return false;
    out.second = rsvp::bcdToBin(b[0] & 0x7F);
    out.minute = rsvp::bcdToBin(b[1] & 0x7F);
    out.hour   = rsvp::bcdToBin(b[2] & 0x3F);
    return true;
}

bool rtc_set(const RtcTime& t) {
    if (!rtc_dev_handle) return false;
    uint8_t w[4] = { 0x04,
        (uint8_t)(rsvp::binToBcd(t.second) & 0x7F),   // bit7=0 clears OS
        rsvp::binToBcd(t.minute),
        rsvp::binToBcd(t.hour) };
    return i2c_master_transmit(rtc_dev_handle, w, 4, kTimeoutMs) == ESP_OK;
}
```

```cpp
void show_setclock() {
    g_screen = SCR_SETCLOCK;
    g_overlay = make_overlay();
    lv_obj_set_flex_flow(g_overlay, LV_FLEX_FLOW_COLUMN);
    lv_obj_set_style_pad_all(g_overlay, 0, 0); lv_obj_set_style_pad_row(g_overlay, 0, 0);
    rsvp::RtcTime t;                               // pre-fill from the RTC, else 12:00 AM
    if (rtc_valid() && rtc_get(t)) {
```
